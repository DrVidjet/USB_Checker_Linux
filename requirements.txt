# Зависимости Python для «Суперапп».
# Установка в виртуальное окружение (из корня проекта):
#   python3 -m venv .venv
#   .venv/bin/pip install -r requirements.txt
#
# PyQt5 и Qt 5.15 ставятся готовыми колёсами manylinux для x86_64,
# компиляция не нужна. PyQt5-sip 12.20.0 требует Python >= 3.10.
# Остальные модули программы (socket, os, urllib, json и др.) входят
# в стандартную библиотеку. Python-клиент для Ollama не нужен:
# программа обращается к её HTTP API через urllib.
PyQt5==5.15.11
PyQt5-Qt5==5.15.19
PyQt5-sip==12.20.0
# Распаковка переносимой сборки Ollama (архив .tar.zst), которую программа
# скачивает сама, если Ollama нет в системе. В Python 3.14 и новее для
# этого есть стандартный модуль compression.zstd, пакет не нужен.
zstandard==0.25.0; python_version < "3.14"
